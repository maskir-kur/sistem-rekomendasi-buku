# Gunakan base image python-slim yang ringan dan hemat RAM
FROM python:3.10-slim

WORKDIR /app

# Install tools esensial jika library python membutuhkan kompilasi C
RUN apt-get update && apt-get install -y --no-install-recommends \
    build-essential \
    && rm -rf /var/lib/apt/lists/*

# Copy file requirements.txt dari project kamu
COPY requirements.txt .

# Install dependensi ke dalam Docker (tanpa menggunakan venv lokal)
RUN pip install --no-cache-dir -r requirements.txt

# Copy seluruh modul ML (termasuk folder scripts dan app.py)
COPY . .

# Abaikan venv lokal bawaan VPS agar tidak masuk ke dalam image Docker
RUN rm -rf venv __pycache__

EXPOSE 8000

# Jalankan Uvicorn dengan file utama 'app' dan variabel FastAPI-mu (asumsi nama variabelnya 'app')
CMD ["uvicorn", "app:app", "--host", "0.0.0.0", "--port", "8000", "--workers", "1"]
